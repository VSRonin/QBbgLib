cmake_minimum_required(VERSION 3.20)
project(QBbgLib LANGUAGES CXX)
set(CMAKE_MODULE_PATH ${CMAKE_CURRENT_SOURCE_DIR}/cmake/modules ${CMAKE_MODULE_PATH})
find_package(QT NAMES Qt6 Qt5 COMPONENTS Core REQUIRED)
option(BUILD_SHARED_LIBS "Build the library as a dynamically linked library" ON)
option(BUILD_EXAMPLES "Build the examples" OFF)
option(BUILD_DOCS "Enables or disables the build of the documentation" OFF)
option(QBBGLIB_OFFLINE "Mocs the Bloomberg interface without linking to the bloomberg libraries" OFF)
option(QBBGLIB_INSTALL "Generate installation target" ON)

include(CTest)
include(GetGitRevisionDescription)
git_describe(GitTagVersion --tags)
string(REGEX MATCH "^[0-9]+\\.[0-9]+\\.[0-9]+.*" VERSION_MATCH "${GitTagVersion}")
if(VERSION_MATCH)
    string(REGEX REPLACE "^([0-9]+)\\..*" "\\1" VERSION_MAJOR "${GitTagVersion}")
    string(REGEX REPLACE "^[0-9]+\\.([0-9]+).*" "\\1" VERSION_MINOR "${GitTagVersion}")
    string(REGEX REPLACE "^[0-9]+\\.[0-9]+\\.([0-9]+).*" "\\1" VERSION_PATCH "${GitTagVersion}")
else()
    set(VERSION_MAJOR 0)
    set(VERSION_MINOR 0)
    set(VERSION_PATCH 0)
endif()
set(VERSION_SHORT "${VERSION_MAJOR}.${VERSION_MINOR}.${VERSION_PATCH}")
message("QBbgLib Version ${VERSION_SHORT}")

if("${CMAKE_SIZEOF_VOID_P}" STREQUAL "4")
    set(qbbglib_IS64BITS OFF)
else()
    set(qbbglib_IS64BITS ON)
endif()
if(qbbglib_IS64BITS)
    set(qbbglib_PlatformDir "x64")
else()
    set(qbbglib_PlatformDir "x86")
endif()

if(CMAKE_BUILD_TYPE MATCHES DEBUG)
    set(BUILD_DOCS OFF)
endif()
add_subdirectory(src)

if(BUILD_EXAMPLES)
    add_subdirectory(examples)
endif()
if(BUILD_TESTING)
    enable_testing()
    add_subdirectory(consoletest)
endif()
if(MODEL_UTILITIES_INSTALL)
    install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE" DESTINATION licenses)
    SET(CPACK_PACKAGE_HOMEPAGE_URL "https://github.com/VSRonin/QBbgLib")
    SET(CPACK_PACKAGE_VERSION_MAJOR ${VERSION_MAJOR})
    SET(CPACK_PACKAGE_VERSION_MINOR ${VERSION_MINOR})
    SET(CPACK_PACKAGE_VERSION_PATCH ${VERSION_PATCH})
    SET(CPACK_PACKAGE_DESCRIPTION_FILE "${CMAKE_CURRENT_SOURCE_DIR}/README.md")
    SET(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE")
    SET(CPACK_PACKAGE_FILE_NAME "QBbgLib-${VERSION_SHORT}-${CMAKE_SYSTEM_NAME}-${CMAKE_CXX_COMPILER_ID}-Qt${QT_VERSION_MAJOR}-${qbbglib_PlatformDir}")
    include(CPack)
endif()
